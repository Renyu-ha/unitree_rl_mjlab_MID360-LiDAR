cmake_minimum_required(VERSION 3.16)
project(go2w_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(GO2W_BUILD_CONTROLLER "Build the Linux DDS/ONNX controller" ON)
if(BUILD_TESTING)
  add_executable(go2w_core_test tests/core_test.cpp)
  target_include_directories(go2w_core_test PRIVATE include)
  add_test(NAME go2w_core COMMAND go2w_core_test)
endif()
if(GO2W_BUILD_CONTROLLER)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "DDS controller targets Linux; use -DGO2W_BUILD_CONTROLLER=OFF for portable core tests")
  endif()
  include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/UnitreeSDK.cmake")
  find_package(Boost REQUIRED COMPONENTS program_options)
  find_package(yaml-cpp REQUIRED)
  find_package(Eigen3 REQUIRED)
  find_package(spdlog REQUIRED)
  find_package(fmt REQUIRED)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    set(ORT_ARCH x64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(ORT_ARCH aarch64)
  else()
    message(FATAL_ERROR "Unsupported ONNX Runtime architecture")
  endif()
  set(ONNXRUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../thirdparty/onnxruntime-linux-${ORT_ARCH}-1.22.0" CACHE PATH "ONNX Runtime distribution")
  add_executable(go2w_ctrl main.cpp src/State_RLBase.cpp)
  target_include_directories(go2w_ctrl PRIVATE include ../../include ${ONNXRUNTIME_DIR}/include)
  target_link_libraries(go2w_ctrl PRIVATE unitree_sdk2 Boost::program_options yaml-cpp Eigen3::Eigen spdlog::spdlog fmt::fmt
    "${ONNXRUNTIME_DIR}/lib/libonnxruntime.so.1.22.0" pthread)
  set_target_properties(go2w_ctrl PROPERTIES BUILD_RPATH "${ONNXRUNTIME_DIR}/lib")
  if(BUILD_TESTING)
    add_executable(go2w_crc_test tests/crc_test.cpp)
    target_link_libraries(go2w_crc_test PRIVATE unitree_sdk2)
    add_test(NAME go2w_crc COMMAND go2w_crc_test)
    add_executable(go2w_channel_test tests/channel_test.cpp)
    target_link_libraries(go2w_channel_test PRIVATE unitree_sdk2)
    add_test(NAME go2w_channel_shutdown COMMAND go2w_channel_test)
  endif()
endif()
